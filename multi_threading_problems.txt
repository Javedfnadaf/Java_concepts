What is thread?

Thread is a small unit of a process used to perform a task.


What is multithreading?

Multithreading is running multiple threads simultaneously.



  start()
👉 Creates new thread


new()
👉 Runs like normal method (no multithreading)

sleep()
👉 Pause thread

join()
👉 Pause thread


class MyThread extends Thread {
    public void run() {
        System.out.println("Thread running...");
    }
}

public class Main {
    public static void main(String[] args) {
        MyThread t1 = new MyThread();
        t1.start();
    }
}

class MyThread extends Thread {
    public void run() {
        try {
            for (int i = 1; i <= 3; i++) {
                System.out.println("Thread: " + i);
                Thread.sleep(1000);
            }
        } catch (InterruptedException e) {
            e.printStackTrace();
        }
    }
}

public class Main {
    public static void main(String[] args) {
        MyThread t1 = new MyThread();
        t1.start();

        try {
            t1.join();  // wait for t1 to finish
        } catch (InterruptedException e) {
            e.printStackTrace();
        }

        System.out.println("Main thread finished");
    }
}

class NumberThread extends Thread {
    public void run() {
        try {
            for (int i = 1; i <= 3; i++) {
                System.out.println(i);
                Thread.sleep(1000); // 1 second delay
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}

class LetterThread extends Thread {
    public void run() {
        try {
            for (char ch = 'A'; ch <= 'C'; ch++) {
                System.out.println(ch);
                Thread.sleep(1000); // 1 second delay
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}

public class Main {
    public static void main(String[] args) {
        NumberThread t1 = new NumberThread();
        LetterThread t2 = new LetterThread();

        t1.start();

        try {
            t1.join(); // wait for t1 to finish
        } catch (InterruptedException e) {
            e.printStackTrace();
        }

        t2.start(); // starts after t1 completes
    }
}
